% One multi-agent round: peers exchange messages and a guard checks provenance.
\documentclass[border=3pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath}
\usepackage{crypto-tikz}
\begin{document}
\begin{tikzpicture}[x=4.2mm, y=-1cm]
  \foreach \i/\name in {0/A,1/B,2/C,3/D} {
    \node[benign, minimum width=13mm] (agent\i) at ({4*\i+1.5},1.0) {Agent \name};
    \draw[wire] ({4*\i+1.5},0) -- (agent\i.north);
  }
  \node[op, anchor=east] at (-0.8,0) {incoming messages};
  \node[attacker, minimum width=15mm] (bad) at (5.5,2.35) {untrusted\nsource};
  \foreach \i in {0,...,3}
    \draw[msg] (bad.north) |- (agent\i.south);
  \node[defense, minimum width=20mm] (guard) at (7.5,3.65) {provenance\nguard};
  \foreach \i in {0,...,3}
    \draw[flow] (agent\i.south) |- (guard.west);
  \node[benign, minimum width=22mm] (relay) at (1.5,4.95) {shared\nsummary};
  \draw[flow] (guard.west) -| (relay.east);
  \node[op, anchor=east] at (-0.8,4.95) {round output};
  \draw[brace] (-0.2,-0.35) -- node[op, above=5pt] {peer state} (15.2,-0.35);
\end{tikzpicture}
\end{document}
